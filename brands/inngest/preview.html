import { Inngest } from "inngest";

const inngest = new Inngest({ id: "acme-app" });

export default inngest.createFunction(
  { id: "onboard-user", retries: 4 },
  { event: "user/signed.up" },
  async ({ event, step }) => {
    // each step is checkpointed & retried on its own
    const customer = await step.run("create-stripe-customer", () =>
      stripe.customers.create({ email: event.data.email })
    );

    const profile = await step.run("enrich-profile", () =>
      clearbit.enrich(event.data.email)
    );

    await step.sleep("await-onboarding", "1h");

    await step.waitForEvent("await-payment", {
      event: "stripe/payment.succeeded",
      timeout: "24h", match: "data.email",
    });

    await step.run("send-welcome-email", () =>
      resend.emails.send({ to: event.data.email })
    );
  }
);

Guardrails
Do
Anchor on a dark, near-black canvas with one hot flame (#F75536) accent — bold and editorial.
Model work as steps on a timeline — a run is an ordered trace, not one opaque spinner.
Set display type oversized in CircularXX; lean into the blunt voice (Unbreakable agents.).
Keep green / amber / blue strictly as step-state signals — functional, never brand chrome.
Show durability out loud — a failed attempt that retries and succeeds, a sleep that resumes.
Don't
Make green (the completed-step color) the brand accent — the brand is flame on black.
Reduce a run to one status — the point is per-step observability.
Use a light/pastel canvas as the primary identity — Inngest is dark and confident.
Set body or code in the display weight, or hide retries — durability is the story.
Invent status colors — reuse the run-dashboard green / amber / blue / red.
