script.js 
import http from 'k6/http';
import { check, sleep } from 'k6';

export const options = {
  stages: [
    { duration: '30s', target: 50 },
    { duration: '1m',  target: 50 },
    { duration: '30s', target: 0 },
  ],
  thresholds: {
    http_req_duration: ['p(95)<500'],
  },
};

export default function () {
  const res = http.get('https://test.k6.io');
  check(res, { 'status is 200': (r) => r.status === 200 });
  sleep(1);
}

Do
Lead with k6 purple (#7D64FF) on lavender-tinted surfaces; use cyan as the data spark.
Show the real k6 vocabulary: http_req_duration, p(95), checks, vus.
Color results by state — green pass, red fail; keep VUs purple, throughput cyan.

Don't
Use pure white/black backgrounds — the canvas is lavender (#F9F8FC) / deep purple (#0E0A1F).
Invent metric names — use k6's actual output fields.
Overuse cyan; it's a spark for data, not a second brand color.
